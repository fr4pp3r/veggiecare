# VeggieCare - Raspberry Pi 5 Smart Plant Monitoring and Control
# Single consolidated dependency set: core + YOLO pest detection.
#
# Install with TWO commands (scripts/setup_rpi5.sh runs both):
#
#     pip install -r requirements.txt
#     pip install --no-deps "ultralytics>=8.4.0"
#
# WHY TWO COMMANDS: ultralytics hard-requires the GUI `opencv-python`,
# which installs into the SAME cv2/ directory as opencv-python-headless
# and needs libGL.so.1 that a headless Pi lacks -> ImportError on import.
# So we install ultralytics with --no-deps and list its dependencies
# explicitly below, keeping the headless build the only OpenCV present.
# Nothing in this project calls cv2.imshow (capture + imwrite + JPEG only).
# ------------------------------------------------------------------

gpiozero>=2.0

# GPIO pin driver - REQUIRED on Linux/ARM. gpiozero>=2.0 ships no backend of
# its own: without one it warns "Falling back from lgpio/rpigpio/pigpio" and
# every GpioRelay is marked unavailable (all relays die silently). lgpio is
# the preferred factory on Pi 5 / kernel 6.x and only needs the `gpio` group.
# Not imported by name anywhere - gpiozero resolves it as a pin factory
# through its entry points.
#
# WHY TWO LINES: upstream lgpio publishes wheels for CPython 3.9-3.12 only, so
# on Python 3.13+ (Raspberry Pi OS Trixie) pip falls back to the sdist and runs
# SWIG - which needs liblgpio-dev and can fail outright. adafruit-lgpio is a
# drop-in `import lgpio` replacement shipping prebuilt aarch64/armv7l wheels for
# 3.13+, so no compiler or SWIG is involved. piwheels is not an option here: it
# does not publish aarch64 wheels at all.
lgpio>=0.2.2.0; python_version<'3.13' or platform_machine not in ('aarch64', 'armv7l')
adafruit-lgpio>=0.2.2.0; sys_platform=='linux' and python_version>='3.13' and platform_machine in ('aarch64', 'armv7l')

# Comment-preserving YAML round-trip for the dashboard Settings page
ruamel.yaml>=0.18

# OpenCV - REQUIRED for the USB camera (capture + MJPEG live view).
# Without it `import cv2` fails inside camera/usb_camera.py, the camera
# silently degrades to simulated frames, and the dashboard reports the
# working webcam as "Not installed". Headless build: no GUI libs.
#
# This must stay the ONLY OpenCV in this file. The GUI build
# (opencv-python, an ultralytics dependency) installs into the same cv2/
# directory and needs libGL.so.1, which a headless Pi usually lacks.
opencv-python-headless>=4.10.0

# Raspberry Pi only (Linux/ARM): required for the soil moisture sensor
# (MCP3008 over SPI). If pip cannot build spidev on the Pi, install via:
#   sudo apt install python3-spidev
spidev>=3.6

# CPU-only torch. Without this line pip resolves the default PyPI wheel,
# which bundles CUDA and is ~454 MB on aarch64 - large enough to exhaust a
# small SD card mid-download (OSError 28). The +cpu build from PyTorch's
# index is a fraction of the size and has no wasted CUDA payload.
--extra-index-url https://download.pytorch.org/whl/cpu

# Pinned as an exact pair: torchvision declares an exact torch requirement,
# and the Pi cannot afford an untested upgrade. `==2.14.1` still matches the
# local version `2.14.1+cpu` (PEP 440 ignores local segments when the
# specifier omits one), so torchvision's own pin is satisfied.
torch==2.14.1+cpu
torchvision==0.29.1+cpu
