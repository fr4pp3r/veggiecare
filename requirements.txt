PyYAML>=6.0

gpiozero>=2.0

# GPIO pin driver — REQUIRED on Linux/ARM. gpiozero>=2.0 ships no backend of
# its own: without one it warns "Falling back from lgpio/rpigpio/pigpio" and
# every GpioRelay is marked unavailable (all relays die silently). lgpio is
# the preferred factory on Pi 5 / kernel 6.x and only needs the `gpio` group.
lgpio>=1.6

# Comment-preserving YAML round-trip for the dashboard Settings page
ruamel.yaml>=0.18

# OpenCV — REQUIRED for the USB camera (capture + MJPEG live view).
# Without it `import cv2` fails inside camera/usb_camera.py, the camera
# silently degrades to simulated frames, and the dashboard reports the
# working webcam as "Not installed".
# Use the headless build on the Pi: no GUI libs, much smaller.
opencv-python-headless>=4.10.0

# Raspberry Pi only (Linux/ARM): required for the soil moisture sensor
# (MCP3008 over SPI). If pip cannot build spidev on the Pi, install via:
#   sudo apt install python3-spidev
spidev>=3.6

pytest>=8.0
ultralytics>=8.3.0
